# Image preservation (Phase W2.0: Shared Image Preservation Foundation) —
# app/ingestion/image_validation.py uses Pillow for real content-signature
# decoding, dimension/pixel-count introspection, and decompression-bomb
# protection on Gmail and direct-upload image attachments (see
# app/email_ingestion/pipeline.py and app/api/routes/project_images.py).
# WhatsApp image attachments are NOT covered by this module -- they go
# through app/whatsapp_ingestion/pipeline.py's own, separate
# _detect_media_type detection, which never imports Pillow and has no
# decompression-bomb guard of its own. Pillow was present in the dev
# environment only incidentally (via an unrelated, otherwise-undeclared
# package) until this phase; declared explicitly now so a clean
# `pip install -r requirements.txt` actually has it — without it, the
# backend fails to import at all (app/api/router.py unconditionally
# registers the image-upload routes). Constrained to the major version
# already tested against (12.3.0) rather than left fully unpinned like
# most of this file, since Pillow's own decompression-bomb API
# (Image.DecompressionBombError/-Warning, see image_validation.py) is the
# exact behavior this phase depends on and verifies.
pillow>=12.3,<13

# Local OCR (Phase W2.1: Local OCR Foundation) — app/ocr/tesseract_provider.py
# uses pytesseract, a thin subprocess wrapper around the real Tesseract OS
# executable. Installing this package does NOT install that executable --
# it must be installed separately on the host (macOS: `brew install
# tesseract`; Debian/Ubuntu: `apt-get install tesseract-ocr`; see the
# README's own "Optional: Local OCR" section). The application itself
# still starts and runs every non-OCR code path normally with Tesseract
# completely absent -- only app/ocr/process_pending.py's own batch command
# needs it, and it exits cleanly (never touching any row) if it's missing.
# Pinned to the exact version its exception/timeout behavior was verified
# against (see tesseract_provider.py's own docstring: TesseractNotFoundError,
# TesseractError, and the plain-RuntimeError timeout path are all confirmed
# against this installed version's actual source, not guessed from docs).
pytesseract==0.3.13
